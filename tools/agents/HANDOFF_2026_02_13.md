## Quick Reference Commands

```bash
# Build headless
cmake --build build-arm64/ --target engine-headless -j$(sysctl -n hw.ncpu)

# Count GL calls (excluding backends, libs, stubs)
grep -rEc "gl[A-Z][a-zA-Z]+\(" rts/ --include="*.cpp" --include="*.h" \
  --exclude-dir="lib" --exclude-dir="headlessStubs" --exclude-dir="glad" \
  --exclude-dir="RHI" --exclude-dir="GL" | awk -F: '{sum+=$2} END {print sum}'

# Top files by GL call count
grep -rEc "gl[A-Z][a-zA-Z]+\(" rts/ --include="*.cpp" --include="*.h" \
  --exclude-dir="lib" --exclude-dir="headlessStubs" --exclude-dir="glad" \
  --exclude-dir="RHI" --exclude-dir="GL" | awk -F: '$2>0' | sort -t: -k2 -nr | head -30

nm build-arm64/rts/CMakeFiles/engine-headless.dir/*.o 2>/dev/null | grep "U _glad_gl"
```
